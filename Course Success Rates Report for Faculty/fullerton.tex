% ------------------------------------------------------------
% Section Heading Styles
% ------------------------------------------------------------

% --- Vertical blue bar for H2/H3 (aligned with text height) ---
\newcommand{\leftbar}{%
  \tikz[baseline]{%
    \node[inner sep=0pt, outer sep=0pt, anchor=base west, yshift=-0.25ex] (bar)
      {\textcolor{FCBlue}{\rule{3pt}{1.25em}}};
  }\hspace{0.6em}%
}

% --- H1 (Header 1): bold, large, left bar ---
\titleformat{\section}[block]
  {\color{FCBlue}\bfseries\LARGE}
  {}{0pt}{\leftbar}



% --- H2 (Header 2): bold, slightly smaller, left bar ---
\titleformat{\subsection}[block]
  {\color{FCBlue}\bfseries\Large}
  {}{0pt}{\leftbar}

% --- H3 (Header 3): bold italic, slightly smaller, left bar ---
\titleformat{\subsubsection}[block]
  {\color{FCBlue}\bfseries\itshape\large}
  {}{0pt}{}

% --- Optional H4–H6 ---
\titleformat{\paragraph}
  {\color{FCDarkGray}\bfseries\normalsize}
  {}{0pt}{}
\titleformat{\subparagraph}
  {\color{FCDarkGray}\normalsize}
  {}{0pt}{}
